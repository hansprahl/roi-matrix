        q2 = st.checkbox("2. Includes clear oversight? (monitoring, stakeholder reporting, course-correction steps?)")
        q2_note = st.text_input("Brief evidence / note:", key="q2_note", disabled=not q2)

        q3 = st.checkbox("3. Supports the four key areas? (strengthens ≥3 of: Community Investment, Healthy Workforce, Quality of Products/Services, Employee Culture & Retention — without severely harming the fourth?)")
        q3_note = st.text_input("Brief evidence / note:", key="q3_note", disabled=not q3)

        q4 = st.checkbox("4. Fits free-market values? (preserves voluntary exchange, informed consent, and competition on genuine value?)")
        q4_note = st.text_input("Brief evidence / note:", key="q4_note", disabled=not q4)

            st.warning("**Conditional Adoption**\nProceed only after addressing the missing condition(s).")
